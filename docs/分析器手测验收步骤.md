## B 组 · 缺模型启动与老算法回归(不需要模型,证明零破坏)

```bash
# B1 重启分析器(加载新二进制;软链接部署,编译产物即生效)
sudo kill $(pgrep -f "Analyzer -f /opt/SVA/config.json"); sleep 2
cd /opt/SVA/server && sudo nohup ./Analyzer -f /opt/SVA/config.json > log.out 2>&1 & sleep 5
# B2 启动日志:应看到两行老模型初始化 + "睡岗模型加载失败(不影响启动)" + "Start Success"

# B3 健康
curl -s http://127.0.0.1:9993/api/health -o /dev/null -w "HTTP %{http_code}\n"
# B4 老算法布控回归(用本机 ffmpeg 假流走通 ZLM → 分析器)
/usr/local/ffmpeg/bin/ffmpeg -re -f lavfi -i "testsrc2=size=640x360:rate=12" -pix_fmt yuv420p \
  -c:v libx264 -preset ultrafast -tune zerolatency -g 24 -f flv rtmp://127.0.0.1:9995/live/mantest & 
sleep 3
curl -s -X POST http://127.0.0.1:9993/api/control/add -d '{"code":"mantest","streamCode":"mantest","streamApp":"live","streamName":"mantest","streamUrl":"rtsp://127.0.0.1:9994/live/mantest","algorithmCode":"on_yolo26n_80","objectCode":"person","objectCodes":["person"],"object_str":"person","regions":[{"id":"r1","primary":true,"points":[0,0,1,0,1,1,0,1]}],"wsEventFps":2,"saveImageEnabled":false,"minInterval":60}'
sleep 10
curl -s http://127.0.0.1:9993/api/health | grep -oE '"detectFrameSent" : [0-9]+'
# B5 清理
curl -s -X POST http://127.0.0.1:9993/api/control/cancel -d '{"code":"mantest"}' ; kill %1
```
